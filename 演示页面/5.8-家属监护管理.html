📄 JS 实时监测

const app = getApp();
Page({
  data: { elder: {}, monitors: [], sosCount: 0 },
  onLoad() {
    const bindId = wx.getStorageSync('bindId');
    this.loadElderInfo(bindId);
    this.startRealtimeMonitor();
  },
  startRealtimeMonitor() {
    // WebSocket 实时推送最新监护数据
    const ws = wx.connectSocket({
      url: 'wss://api.eldercare.com/ws/monitor'
    });
    ws.onMessage(res => {
      const data = JSON.parse(res.data);
      if (data.type === 'vitals') {
        this.setData({ monitors: data.payload });
      }
      if (data.type === 'alert') {
        wx.showModal({
          title: '⚠️ 异常预警',
          content: data.msg,
          confirmText: '查看详情'
        });
      }
    });
  },
  onUnload() { wx.closeSocket(); }
});
