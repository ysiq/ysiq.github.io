📄 HealthDataMapper.xml - 数据查询
<!-- 健康数据分页查询 -->
<select id="selectHealthDataList"
        resultType="com.elder.entity.HealthData">
  SELECT h.*, u.real_name AS elderName
  FROM health_data h
  LEFT JOIN user u ON h.user_id = u.user_id
  <where>
    <if test="elderId != null">
      AND h.user_id = #{elderId}
    </if>
    <if test="type != null and type != ''">
      AND h.data_type = #{type}
    </if>
    <if test="startTime != null">
      AND h.create_time >= #{startTime}
    </if>
    <if test="endTime != null">
      AND h.create_time <= #{endTime}
    </if>
    <if test="status != null">
      AND h.status = #{status}
    </if>
  </where>
  ORDER BY h.create_time DESC
</select>
