📄 消息推送服务 PushService.java
@Service
public class PushService {

    @Autowired private WxTemplateMsgSender wxSender;
    @Autowired private SmsSender smsSender;

    public void pushHealthAlert(HealthAlertDTO alert) {
        // 1. 查询绑定家属
        List<FamilyBind> families = familyBindMapper
                .selectByElderId(alert.getElderId());
        // 2. 组装微信模板消息
        WxTemplateMsg msg = new WxTemplateMsg();
        msg.setTemplateId("TEMPLATE_HEALTH_ALERT");
        msg.setPage("/pages/health/detail");
        msg.setData(Map.of(
            "thing1", Map.of("value", "健康异常预警"),
            "thing2", Map.of("value", alert.getAlertMsg())
        ));
        // 3. 逐个推送
        for (FamilyBind fb : families) {
            msg.setToUser(fb.getFamilyOpenId());
            wxSender.send(msg);
        }
        // 4. 记录推送日志
        pushLogMapper.insert(new PushLog(alert));
    }
}
