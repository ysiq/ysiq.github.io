📄 JS 获取健康数据

Page({
  data: {
    range: 'week',
    metrics: {},
    trend: [],
    records: []
  },
  onLoad() { this.loadData(); },
  async loadData() {
    wx.showLoading({ title: '加载中...' });
    try {
      const res = await wx.request({
        url: `https://api.eldercare.com/api/health/stats`,
        data: { range: this.data.range }
      });
      if (res.data.code === 200) {
        this.setData({
          metrics: res.data.data.metrics,
          trend: res.data.data.trend,
          records: res.data.data.records
        });
      }
    } catch (err) {
      wx.showToast({ title: '数据加载失败', icon: 'error' });
    } finally { wx.hideLoading(); }
  },
  switchRange(e) {
    this.setData({ range: e.currentTarget.dataset.r });
    this.loadData();
  }
});

📄 Java 统计数据接口

@GetMapping("/stats")
public Result getHealthStats(@RequestParam String range) {
    Integer userId = CurrentUser.getId();
    LocalDateTime start = switch (range) {
        case "week"  -> LocalDateTime.now().minusDays(7);
        case "month" -> LocalDateTime.now().minusDays(30);
        case "quarter" -> LocalDateTime.now().minusDays(90);
        default -> LocalDateTime.now().minusDays(7);
    };
    List<HealthData> list = service.findByUserAndTime(userId, start);
    Map<String, Object> stats = service.calculateStats(list);
    return Result.success(stats);
}
