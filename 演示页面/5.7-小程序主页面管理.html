📄 JS 首页逻辑

Page({
  data: { userName: '', vitals: [], menus: [], messages: [] },
  onLoad() {
    const user = wx.getStorageSync('userInfo');
    this.setData({ userName: user.realName });
    this.loadVitals();
    this.loadMessages();
  },
  async loadVitals() {
    const res = await wx.request({
      url: 'https://api.eldercare.com/api/home/vitals'
    });
    if (res.data.code === 200) {
      this.setData({ vitals: res.data.data });
    }
  },
  goPage(e) {
    wx.navigateTo({ url: e.currentTarget.dataset.url });
  }
});
